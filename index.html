import numpy as np

def sigmoid(z):
    return 1 / (1 + np.exp(-z))

# 2 -> h1 -> h2 -> 1 的 MLP，二分类
def forward(x, params):
    a = x
    for W, b in params[:-1]:
        a = np.tanh(W @ a + b)              # 隐藏层（此处以 tanh 为例）
    W, b = params[-1]
    return sigmoid(W @ a + b)               # 输出层 sigmoid

def train(X, y, params, lr):
    # 前向
    a = X
    cache = [a]
    for W, b in params[:-1]:
        a = np.tanh(a @ W.T + b); cache.append(a)
    W, b = params[-1]
    logit = a @ W.T + b; prob = sigmoid(logit)
    # 损失：二元交叉熵
    loss = -(y * np.log(prob) + (1 - y) * np.log(1 - prob)).mean()
    # 反向传播（链式法则）→ 更新每个 W、b
    # ...（与本页面 JS 逻辑一致）
    return loss

提示：切换数据集会按经验自动设置激活函数（XOR 用 Tanh、其余用 ReLU）。同心圆 / 双螺旋需要更多轮训练才能收敛，可加大「训练轮数」。
